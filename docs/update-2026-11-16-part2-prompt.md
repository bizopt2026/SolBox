```js
function cwLapSplit(sec){var eLo=cwZ("e")[0],out=[];
  function longQ(x){return x&&!x.intervalGroup&&x.sectionType===2&&x.targetType===1&&x.intensityType===2&&x.targetValue>=2000&&(x.intensityValueStart+x.intensityValueEnd)/2<eLo}
  // repeats with long quality reps (e.g. MP 7K x3 + 1K float): unroll so each rep is split into km laps, recovery stays between reps
  var flat=[];sec.forEach(function(x){if(x.intervalGroup&&x.sets.some(longQ)&&x.repeats*x.sets.length<=12){for(var r=0;r<x.repeats;r++)x.sets.forEach(function(y){var c={};for(var k in y)c[k]=y[k];if(!longQ(c))c.sectionType=c.sectionType===3?3:2;flat.push(c)})}else flat.push(x)});sec=flat;
  sec.forEach(function(x){if(x.intervalGroup||x.sectionType!==2||x.targetType!==1||x.intensityType!==2||x.targetValue<2000){out.push(x);return}
    var avg=(x.intensityValueStart+x.intensityValueEnd)/2;if(avg>=eLo){out.push(x);return}
    var lap=x.targetValue<=20000?1000:2000,n=Math.min(20,Math.floor(x.targetValue/lap)),rem=x.targetValue-n*lap;if(n<2){out.push(x);return}
    var one={};for(var k in x)one[k]=x[k];one.targetValue=lap;out.push({intervalGroup:true,repeats:n,sets:[one]});
    if(rem>=100){var r={};for(var k2 in x)r[k2]=x[k2];r.targetValue=rem;out.push(r)}});
  return out}
```
